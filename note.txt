        app = initializeApp(firebaseConfig);

    async function resolveRound() {
        let winningNumber;
        const rigVal = localStorage.getItem('92team_rig_number');
        if (rigVal !== null && rigVal !== undefined && rigVal !== 'random') {
            winningNumber = parseInt(rigVal);
            localStorage.removeItem('92team_rig_number');
        } else {
            winningNumber = Math.floor(Math.random() * 10);
        }

        const winningColors = getNumberColors(winningNumber);
        const currentPeriod = periodCounter.toString();

        await window.saveGameHistoryToCloud({
            period: currentPeriod,
            number: winningNumber,
            colors: winningColors,
            timestamp: Date.now()
        });

        window.cloudBets.forEach(async b => {
            if (b.period === currentPeriod && b.status === 'pending') {
                let isWin = false;
                const winSize = winningNumber >= 5 ? 'Big' : 'Small';
                const winColorStr = winningColors.join('/');

                if (b.targetLabel.includes('Number') && b.targetLabel.includes(winningNumber.toString())) isWin = true;
                if (b.targetLabel === 'Green' && winColorStr.includes('Green')) isWin = true;
                if (b.targetLabel === 'Red' && winColorStr.includes('Red')) isWin = true;
                if (b.targetLabel === 'Violet' && winColorStr.includes('Violet')) isWin = true;
                if (b.targetLabel === 'Big' && winSize === 'Big') isWin = true;
                if (b.targetLabel === 'Small' && winSize === 'Small') isWin = true;

                if (isWin) {
                    const payout = b.amount * b.multiplier;
                    const u = window.cloudUsers[b.userId];
                    if (u) {
                        await window.updateUserBalanceInCloud(b.userId, u.balance + payout);
                    }
                }
            }
        });
    }

        await window.updateTransactionStatusInCloud(txId, action === 'approve' ? 'approved' : 'rejected');
        if (action === 'approve' && type === 'deposit') {
            const targetUser = window.cloudUsers[userId];
            if (targetUser) {
                await window.updateUserBalanceInCloud(userId, targetUser.balance + amount);
            }
        } else if (action === 'reject' && type === 'withdraw') {
            const targetUser = window.cloudUsers[userId];
            if (targetUser) {
                await window.updateUserBalanceInCloud(userId, targetUser.balance + amount);
            }
        }
        showToast(`Transaction ${action}d successfully!`, 'success');
    };

    window.saveAdminRig = function() {
        const val = document.getElementById('admin-rig-select').value;
        localStorage.setItem('92team_rig_number', val);
        showToast(`Next round winning result set to: ${val}`, 'success');
    };

    window.handleAddNewGame = async function() {
        const title = document.getElementById('admin-new-game-title').value.trim();
        const code = document.getElementById('admin-new-game-code').value.trim();

        if (!title || !code) {
            showToast('Please enter game title and HTML/JS code.', 'error');
            return;
        }

        await window.addCustomGameToCloud({ title, code });
        document.getElementById('admin-new-game-title').value = '';
        document.getElementById('admin-new-game-code').value = '';
        showToast('New custom game published successfully!', 'success');
    };

    function refreshCustomGamesList() {
        const listEl = document.getElementById('custom-games-list');
        if (!listEl) return;
        if (!window.cloudCustomGames || window.cloudCustomGames.length === 0) {
            listEl.innerHTML = `<div class="bg-cardBg card-border p-6 rounded-2xl text-center text-gray-400 text-xs">No custom games published yet. Admin can add games from Admin Panel.</div>`;
            return;
        }

        listEl.innerHTML = window.cloudCustomGames.map((g, idx) => `
            <div class="bg-cardBg card-border p-4 rounded-2xl flex items-center justify-between">
                <div>
                    <h4 class="font-bold text-white text-sm">${g.title}</h4>
                    <p class="text-[11px] text-gray-400">Custom HTML Game</p>
                </div>
                <button onclick="playCustomGame('${g.id}')" class="px-4 py-2 rounded-xl gold-gradient text-black font-bold text-xs shadow">Play Now</button>
            </div>
        `).join('');
    }

    window.playCustomGame = function(gameId) {
        const game = window.cloudCustomGames.find(g => g.id === gameId);
        if (!game) return;
        document.getElementById('playGameTitle').textContent = game.title;
        document.getElementById('playGameContainer').innerHTML = game.code;
        document.getElementById('playCustomGameModal').classList.remove('hidden');
    };

    window.closePlayCustomGame = function() {
        document.getElementById('playCustomGameModal').classList.add('hidden');
        document.getElementById('playGameContainer').innerHTML = '';
    };

    window.refreshAllUI = function() {
        const user = window.getCurrentUser();
        if (!user) return;

        const balFormatted = user.balance.toFixed(2);
        document.getElementById('header-balance').textContent = balFormatted;
        document.getElementById('dash-main-bal').textContent = balFormatted;
        document.getElementById('wallet-main-balance').textContent = balFormatted;

        const allUsersArr = Object.values(window.cloudUsers);
        const myTeam = allUsersArr.filter(u => u.referredBy === user.inviteCode);
        document.getElementById('promo-team-count').textContent = myTeam.length;
        document.getElementById('promo-commission').textContent = (user.commission || 0).toFixed(0);
        document.getElementById('user-invite-code').textContent = user.inviteCode;
        const baseUrl = window.location.href.split('?')[0].split('#')[0];
        document.getElementById('user-invite-link').value = `${baseUrl}?invitationCode=${user.inviteCode}`;

        const histBody = document.getElementById('game-history-body');
        if (histBody) {
            histBody.innerHTML = window.cloudGameHistory.slice(0, 10).map(row => {
                const numColor = row.colors?.includes('Green') ? 'text-vipGreen' : 'text-vipRed';
                const dots = row.colors?.map(c => `<span class="inline-block w-3 h-3 rounded-full ${c === 'Green' ? 'bg-vipGreen' : c === 'Red' ? 'bg-vipRed' : 'bg-vipViolet'}"></span>`).join(' ') || '';
                return `
                    <tr>
                        <td class="py-2.5 font-mono text-gray-300">${row.period}</td>
                        <td class="py-2.5 text-center font-extrabold text-base ${numColor}">${row.number}</td>
                        <td class="py-2.5 text-right space-x-1">${dots}</td>
                    </tr>
                `;
            }).join('');
        }

        const myBetsList = document.getElementById('my-bets-list');
        if (myBetsList) {
            const myBets = window.cloudBets.filter(b => b.userId === user.phone);
            myBetsList.innerHTML = myBets.length === 0 ? '<p class="text-gray-500 text-xs text-center py-4">No bets placed yet.</p>' : myBets.slice(0, 15).map(b => `
                <div class="flex items-center justify-between bg-darkBg p-2.5 rounded-xl border border-gray-800 text-xs">
                    <div>
                        <p class="font-bold text-white">${b.targetLabel} <span class="text-gray-500 font-mono text-[10px]">(#${b.period})</span></p>
                        <p class="text-[11px] text-gray-400">Bet: Rs. ${b.amount}</p>
                    </div>
                    <span class="text-goldPrimary font-semibold uppercase">${b.status}</span>
                </div>
            `).join('');
        }

        const txContainer = document.getElementById('user-tx-history');
        if (txContainer) {
            const myTxs = window.cloudTransactions.filter(t => t.userId === user.phone);
            txContainer.innerHTML = myTxs.length === 0 ? '<p class="text-gray-500 text-center py-3">No transactions found.</p>' : myTxs.map(t => `
                <div class="bg-darkBg p-3 rounded-xl border border-gray-800 flex items-center justify-between">
                    <div>
                        <p class="font-bold text-white uppercase">${t.type} &bull; ${t.method}</p>
                        <p class="text-[11px] text-gray-400">${t.accountInfo}</p>
                    </div>
                    <div class="text-right">
                        <p class="font-extrabold text-sm text-white">Rs. ${t.amount}</p>
                        <span class="text-[11px] font-bold uppercase ${t.status === 'approved' ? 'text-vipGreen' : t.status === 'rejected' ? 'text-vipRed' : 'text-goldPrimary'}">${t.status}</span>
                    </div>
                </div>
            `).join('');
        }

        const pendingDeps = window.cloudTransactions.filter(t => t.type === 'deposit' && t.status === 'pending').length;
        const pendingWiths = window.cloudTransactions.filter(t => t.type === 'withdraw' && t.status === 'pending').length;
        document.getElementById('admin-stat-users').textContent = allUsersArr.length;
        document.getElementById('admin-stat-dep').textContent = pendingDeps;
        document.getElementById('admin-stat-with').textContent = pendingWiths;

        const adminReqList = document.getElementById('admin-requests-list');
        if (adminReqList) {
            adminReqList.innerHTML = window.cloudTransactions.length === 0 ? '<p class="text-gray-500 text-center py-3">No pending requests.</p>' : window.cloudTransactions.map(t => {
                const actionBtns = t.status === 'pending' ? `
                    <div class="flex gap-2 mt-2">
                        <button onclick="adminHandleTx('${t.id}', 'approve', '${t.userId}', ${t.amount}, '${t.type}')" class="flex-1 py-1.5 rounded-lg bg-vipGreen text-white font-bold text-[11px]">Approve</button>
                        <button onclick="adminHandleTx('${t.id}', 'reject', '${t.userId}', ${t.amount}, '${t.type}')" class="flex-1 py-1.5 rounded-lg bg-red-600 text-white font-bold text-[11px]">Reject</button>
                    </div>
                ` : `<span class="inline-block mt-1 text-[10px] uppercase font-bold ${t.status === 'approved' ? 'text-vipGreen' : 'text-red-400'}">${t.status}</span>`;

                return `
                    <div class="bg-darkBg p-3 rounded-xl border border-gray-800">
                        <div class="flex justify-between items-start">
                            <div>
                                <span class="text-[10px] px-2 py-0.5 rounded font-bold uppercase ${t.type === 'deposit' ? 'bg-emerald-950 text-emerald-400' : 'bg-red-950 text-red-400'}">${t.type}</span>
                                <span class="font-bold text-white ml-1">${t.userName}</span>
                                <p class="text-[11px] text-gray-400 mt-1">${t.accountInfo}</p>
                            </div>
                            <span class="font-extrabold text-goldPrimary text-sm">Rs. ${t.amount}</span>
                        </div>
                        ${actionBtns}
                    </div>
                `;
            }).join('');
        }
    };
